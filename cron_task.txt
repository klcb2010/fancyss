# BEGIN CUSTOM CRON TASK
# ======================= 清理旧任务，防止重复 =======================
for task in frpc_guard rclone_guard clean_logs refresh_ddns rclone_cache_clean ss_update_daily daily_check; do
    cru d $task 2>/dev/null
done

# ======================= 1.0 注入定时守护任务 =======================
cru a frpc_guard "*/10 * * * * [ -z \"\$(pidof frpc)\" ] && /bin/sh /jffs/scripts/frpc_start.sh"
cru a rclone_guard "*/20 * * * * [ -z \"\$(pidof rclone)\" ] && /bin/sh /jffs/scripts/rclone_webdav.sh"
cru a refresh_ddns "30 3 */5 * * /jffs/scripts/refresh_ddns.sh >> /jffs/scripts/ddns_refresh.log 2>&1"
cru a ss_update_daily "30 4 * * * sh /jffs/.koolshare/scripts/ss_update.sh update >> /jffs/scripts/ss_update.log 2>&1"

# ======================= 2.0 注入清理任务 =======================
mkdir -p /jffs/scripts
cru a clean_logs "0 3 * * * /jffs/scripts/clean_logs.sh >> /jffs/scripts/clean_logs.txt 2>&1"
cru a rclone_cache_clean "0 5 * * * [ -d /tmp/mnt/SF/entware/var/cache/rclone ] && rm -rf /tmp/mnt/SF/entware/var/cache/rclone/*"
# 每天凌晨 5:30 运行，并将脚本本身的报错也记录到系统日志或黑洞
cru a daily_check "30 5 * * * /bin/sh /jffs/scripts/check_status.sh >> /jffs/scripts/check_status.log 2>&1"

# ======================= 3.0 运行后台服务守护 (PID 锁逻辑) =======================
# rclone_webdav 守护
RCLONE_PID="/tmp/rclone_webdav.pid"
if [ -f "$RCLONE_PID" ] && kill -0 $(cat "$RCLONE_PID") 2>/dev/null; then
    echo "$(date '+%Y-%m-%d %H:%M:%S') [跳过] rclone_webdav.sh 正在运行" >> /tmp/rclone_webdav.log
else
    pkill -f rclone_webdav.sh 2>/dev/null
    /bin/sh /jffs/scripts/rclone_webdav.sh >> /tmp/rclone_webdav.log 2>&1 &
    echo $! > "$RCLONE_PID"
fi

#!/bin/sh
# =================================================================
# GT-AX6000 v6.3 灵动修复版 (解决 --to-ports 报错)
# =================================================================

# 1. 端口安全获取：增加三层保护逻辑
# 第一层：从 nvram 获取；第二层：从重定向端口获取；第三层：硬编码默认 3333
FINAL_PORT=$(nvram get ss_status_proxy_port)
[ -z "$FINAL_PORT" ] && FINAL_PORT=$(nvram get ss_redir_port)
[ -z "$FINAL_PORT" ] && FINAL_PORT="3333"

# 2. 灵动域名拦截 (插入到 dnsmasq 配置)
ipset -N ai_bypass hash:net 2>/dev/null
# 修正：采用 -p 保证创建目录不报错，强制覆盖确保配置最新
mkdir -p /jffs/configs/dnsmasq.d
cat << EOF > /jffs/configs/dnsmasq.d/ai_proxy.conf
# 抓取 AI 域名 IP 自动填入内核 ipset
server=/openai.com/127.0.0.1#5353
ipset=/openai.com/ai_bypass
server=/chatgpt.com/127.0.0.1#5353
ipset=/chatgpt.com/ai_bypass
server=/anthropic.com/127.0.0.1#5353
ipset=/anthropic.com/ai_bypass
server=/x.com/127.0.0.1#5353
ipset=/x.com/ai_bypass
EOF

# 3. 稳健注入 iptables (针对刚才报错的修复)
# 这里用了 "$FINAL_PORT" 确保即使是空格也不会语法错误
iptables -t nat -D PREROUTING -p tcp -m set --match-set ai_bypass dst -j REDIRECT --to-ports "$FINAL_PORT" 2>/dev/null
iptables -t nat -I PREROUTING -p tcp -m set --match-set ai_bypass dst -j REDIRECT --to-ports "$FINAL_PORT"

# 4. 8181 IPv6 管理端口放行
ip6tables -C INPUT -p tcp --dport 8181 -j ACCEPT 2>/dev/null || ip6tables -I INPUT 1 -p tcp --dport 8181 -j ACCEPT

# 5. IPv6 0分止损 (彻底解决 test-ipv6 的问题)
# 只要不强制 DNAT 53 端口，你的 10/10 分就稳了
while ip6tables -t nat -D PREROUTING -p udp --dport 53 -j DNAT 2>/dev/null; do :; done

echo "灵动版 v6.3 部署成功，重定向端口: $FINAL_PORT"
