# 阻止 外部IPv4 访问 2525，只允许 IPv6
# IPv6 防火墙规则 - 只暴露必要端口给外部，内部 LAN 全放行
# 适用于只有动态公网 IPv6 的场景，配合 asuscomm.com DDNS 使用

# 1. 允许已建立和相关的连接（必须放行，否则连接会断）
ip6tables -A INPUT -m conntrack --ctstate ESTABLISHED,RELATED -j ACCEPT
ip6tables -A FORWARD -m conntrack --ctstate ESTABLISHED,RELATED -j ACCEPT

# 2. 允许 ICMPv6（IPv6 必须，否则 ping、邻居发现等全失效）
ip6tables -A INPUT -p ipv6-icmp -j ACCEPT
ip6tables -A FORWARD -p ipv6-icmp -j ACCEPT

# 3. 内网（br0）访问 2525 端口完全放行（类似你原来的 IPv4 规则）
ip6tables -C INPUT -p tcp -i br0 --dport 2525 -j ACCEPT 2>/dev/null || \
ip6tables -I INPUT 1 -p tcp -i br0 --dport 2525 -j ACCEPT

# 4. 外部 IPv6 放行 2525 端口（远程访问核心端口，不限来源）
ip6tables -C INPUT -p tcp --dport 2525 -j ACCEPT 2>/dev/null || \
ip6tables -I INPUT 2 -p tcp --dport 2525 -j ACCEPT

# 5. 外部 IPv6 放行华硕 Web GUI 默认 HTTPS 端口 8443（可选，如果你也用这个端口）
ip6tables -C INPUT -p tcp --dport 8443 -j ACCEPT 2>/dev/null || \
ip6tables -I INPUT 3 -p tcp --dport 8443 -j ACCEPT

# 6. 可选：如果你还开了 SSH，可以加这一行（生产环境慎开）
# ip6tables -I INPUT 4 -p tcp --dport 22 -j ACCEPT

# 7. 确保 fancyss 的 ipv6only_list6（DDNS 域名）流量不被 nat 干扰
if ipset list ipv6only_list6 >/dev/null 2>&1; then
    ip6tables -t nat -C PREROUTING -m set --match-set ipv6only_list6 dst -j ACCEPT 2>/dev/null || \
    ip6tables -t nat -I PREROUTING 1 -m set --match-set ipv6only_list6 dst -j ACCEPT
else
    logger -t "fancyss-custom" "警告: ipv6only_list6 尚未创建，nat规则延迟生效"
fi

echo_date "======================= 华硕DDNS防火墙规则已启用 ======================"
