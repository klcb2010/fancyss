#!/bin/sh
# =================================================================
# GT-AX6000 深度自检脚本 v7.1 (针对 3.5.8 AAAA 过滤优化)
# =================================================================

echo "==================== [ GT-AX6000 环境监控 ] ===================="

# 1. 核心端口与规则堆叠检查
check_rule() {
    local cmd="$1"
    local port="$2"
    local count=$($cmd -L INPUT -n --line-numbers | grep "$port" | wc -l)
    if [ "$count" -eq 1 ]; then
        echo "✅ $port 端口: 正常 (1条规则)"
    elif [ "$count" -gt 1 ]; then
        echo "❌ $port 端口: 堆叠堆死! (当前: $count 条，建议重启 nat-start)"
    else
        echo "❌ $port 端口: 缺失 (防火墙未放行)"
    fi
}

check_rule "iptables" "2525"
check_rule "ip6tables" "8181"

# 2. 灵动分流 (ipset) 活跃度监控
AI_IP_COUNT=$(ipset list ai_bypass 2>/dev/null | grep -E "^[0-9]" | wc -l)
if [ "$AI_IP_COUNT" -gt 0 ]; then
    echo "✅ IPSet (ai_bypass): 激活 (捕获 $AI_IP_COUNT 个 IP)"
else
    echo "⚠️  IPSet (ai_bypass): 空闲 (尚未触发灵动分流)"
fi

# 3. IPv6 AAAA 过滤坑位精准打击 (核心检查)

# A. 物理链路测试
if ping6 -c 1 -w 2 240c::6666 >/dev/null 2>&1; then
    echo "✅ IPv6 物理链路: 畅通"
else
    echo "❌ IPv6 物理链路: 阻塞 (请检查 WAN 口 IPv6)"
fi

# B. 插件 AAAA 过滤开关测试 (模拟请求 OpenAI)
# 如果返回空或者 0.0.0.0，说明 3.5.8 的那个大坑还没填好
AAAA_RES=$(nslookup -query=AAAA google.com 127.0.0.1 2>/dev/null | grep "has IPv6 address")
if [ -z "$AAAA_RES" ]; then
    echo "🚨 警报: 检测到 AAAA 记录被拦截!"
    echo "   >>> 请检查插件面板: '过滤代理 AAAA 记录' 必须关闭！"
else
    echo "✅ AAAA 解析: 正常 "
fi

# 4. 0分元凶：残留的 ip6tables DNAT 检查
V6_DNAT_COUNT=$(ip6tables -t nat -L PREROUTING -n | grep "DNAT" | grep "53" | wc -l)
if [ "$V6_DNAT_COUNT" -gt 0 ]; then
    echo "🚨 发现残留劫持: $V6_DNAT_COUNT 条 (会导致 test-ipv6 报 0 分)"
    echo "   >>> 解决: 请再次运行 v6.5 nat-start 脚本进行暴力清理"
fi

echo "================================================================"
