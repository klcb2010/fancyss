# ======================= 3.0 后台服务温柔启动 (存在即跳过) =======================
start_service_soft() {
    local _script="$1"
    local _name=$(basename "$_script")
    
    # 第一重检查：检查 ps 进程列表里是否已经有这个脚本在跑
    if ps -w | grep -v grep | grep -q "$_name"; then
        return 0
    fi
    
    # 第二重检查：针对 rclone 等二进制程序的检查 (如果脚本已经在跑但没拉起程序，这里补一下)
    # 这里直接运行脚本，脚本内部建议也带上同样的“存在即跳过”逻辑
    /bin/sh "$_script" >/dev/null 2>&1 &
}

# 1. 直接通过变量获取格式化时间，规避函数定义的括号冲突

#!/bin/sh
LOG_TIME=$(date '+%Y-%m-%d %H:%M:%S')

# 1. 定义清理函数（增加强制清理逻辑）
flush_custom_rules() {
    local p=$1
    local proto=$2
    # 只要规则还在，就一直删，直到报错为止
    while $proto -D INPUT -p tcp --dport $p -j ACCEPT 2>/dev/null; do :; done
    while $proto -D INPUT -p tcp --dport $p -m state --state NEW -m recent --set 2>/dev/null; do :; done
    # 清理所有带 hit_count 的规则（不管它是 8 还是 10，不管它是 LOG 还是 DROP）
    # 注意：这里可能需要多运行几次或者手动清理残留的复杂 LOG 规则
}

# 2. 强力清理（建议手动执行 iptables -F INPUT 如果你不怕断连，或者按下面精准清理）
for port in 2525 8181; do
    flush_custom_rules $port iptables
    flush_custom_rules $port ip6tables
done

echo "$LOG_TIME 清理完成，开始应用优化规则..."

# ---------------------------------------------------------
# 3. 核心规则应用 (使用 -A 顺序控制)
# ---------------------------------------------------------

# 【A部分：放行已有连接】（最优先级）
iptables -I INPUT 1 -m state --state RELATED,ESTABLISHED -j ACCEPT
ip6tables -I INPUT 1 -m state --state RELATED,ESTABLISHED -j ACCEPT

# 【B部分：WebDAV 8181 加固】
# 考虑到 WebDAV 会有缩略图和并发，我们将阈值放宽到 20次/60秒
for proto in "iptables" "ip6tables"; do
    # 达到 20 次先记日志
    $proto -A INPUT -p tcp --dport 8181 -m state --state NEW -m recent --rcheck --seconds 60 --hitcount 20 -m limit --limit 1/min -j LOG --log-prefix "8181-BLOCK: "
    # 达到 20 次直接踢掉
    $proto -A INPUT -p tcp --dport 8181 -m state --state NEW -m recent --rcheck --seconds 60 --hitcount 20 -j DROP
    # 记录并放行
    $proto -A INPUT -p tcp --dport 8181 -m state --state NEW -m recent --set
    $proto -A INPUT -p tcp --dport 8181 -j ACCEPT
done

# 【C部分：DDNS 2525 加固】
# 这个端口请求频率低，可以严格一点，维持 8次/60秒
for proto in "iptables" "ip6tables"; do
    $proto -A INPUT -p tcp --dport 2525 -m state --state NEW -m recent --rcheck --seconds 60 --hitcount 8 -m limit --limit 1/min -j LOG --log-prefix "2525-BLOCK: "
    $proto -A INPUT -p tcp --dport 2525 -m state --state NEW -m recent --rcheck --seconds 60 --hitcount 8 -j DROP
    $proto -A INPUT -p tcp --dport 2525 -m state --state NEW -m recent --set
    $proto -A INPUT -p tcp --dport 2525 -j ACCEPT
done

echo "$LOG_TIME 规则重载成功！"


# --- 第三部分：AI 分流补漏 (OpenAI + Grok 纯净稳定版) ---
# 清空旧规则
while iptables -t nat -D PREROUTING -p tcp -m set --match-set ai_bypass dst -j REDIRECT 2>/dev/null; do :; done

# 重建纯净 ipset（无残留、无脏数据）
ipset -F ai_bypass 2>/dev/null
ipset -X ai_bypass 2>/dev/null
ipset -N ai_bypass hash:net 2>/dev/null

# 2026最新精准IP：OpenAI + Grok 专用（不误杀任何网站）
AI_NETS="131.0.72.0/22 104.244.42.0/24 192.133.76.0/22 199.59.148.0/22"
for net in $AI_NETS; do
    ipset -! add ai_bypass $net 2>/dev/null
done

# 端口容错（自动获取代理端口，失败用3333兜底）
RAW_PORT=$(nvram get ss_status_proxy_port | tr -d '[:space:]')
if [ -z "$RAW_PORT" ] || ! echo "$RAW_PORT" | grep -q '^[0-9]\+$'; then
    FINAL_PORT="3333"
else
    FINAL_PORT="$RAW_PORT"
fi

# 应用分流规则（只对 AI 网段生效）
iptables -t nat -A PREROUTING -p tcp -m set --match-set ai_bypass dst -j REDIRECT --to-ports "$FINAL_PORT"

echo "$LOG_TIME ======================= ✅ AI分流已挂载 ============================ "
echo "$LOG_TIME ======================= 自定义防火墙完毕 ============================ "
