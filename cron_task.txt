# BEGIN CUSTOM CRON TASK
# ======================= 清理旧任务，防止重复 =======================
for task in frpc_guard rclone_guard clean_logs refresh_ddns rclone_cache_clean ss_update_daily daily_check; do
    cru d $task 2>/dev/null
done

# ======================= 1.0 注入定时守护任务 =======================
cru a frpc_guard "*/10 * * * * [ -z \"\$(pidof frpc)\" ] && /bin/sh /jffs/scripts/frpc_start.sh"
cru a rclone_guard "*/20 * * * * [ -z \"\$(pidof rclone)\" ] && /bin/sh /jffs/scripts/rclone_webdav.sh"
cru a refresh_ddns "30 3 */5 * * /jffs/scripts/refresh_ddns.sh >> /jffs/scripts/ddns_refresh.log 2>&1"
cru a ss_update_daily "30 4 * * * sh /jffs/.koolshare/scripts/ss_update.sh update >> /jffs/scripts/ss_update.log 2>&1"

# ======================= 2.0 注入清理任务 =======================
mkdir -p /jffs/scripts
cru a clean_logs "0 3 * * * /jffs/scripts/clean_logs.sh >> /jffs/scripts/clean_logs.txt 2>&1"
cru a rclone_cache_clean "0 5 * * * [ -d /tmp/mnt/SF/entware/var/cache/rclone ] && rm -rf /tmp/mnt/SF/entware/var/cache/rclone/*"
# 每天凌晨 5:30 运行，并将脚本本身的报错也记录到系统日志或黑洞
cru a daily_check "30 5 * * * /bin/sh /jffs/scripts/check_status.sh >> /jffs/scripts/check_status.log 2>&1"

# ======================= 3.0 运行后台服务守护 (PID 锁逻辑) =======================
# rclone_webdav 守护
RCLONE_PID="/tmp/rclone_webdav.pid"
if [ -f "$RCLONE_PID" ] && kill -0 $(cat "$RCLONE_PID") 2>/dev/null; then
    echo "$(date '+%Y-%m-%d %H:%M:%S') [跳过] rclone_webdav.sh 正在运行" >> /tmp/rclone_webdav.log
else
    pkill -f rclone_webdav.sh 2>/dev/null
    /bin/sh /jffs/scripts/rclone_webdav.sh >> /tmp/rclone_webdav.log 2>&1 &
    echo $! > "$RCLONE_PID"
fi

# DDNS_SSH_helper 守护
DDNS_PID="/tmp/ddns_ssh_helper.pid"
if [ -f "$DDNS_PID" ] && kill -0 $(cat "$DDNS_PID") 2>/dev/null; then
    echo "$(date '+%Y-%m-%d %H:%M:%S') [跳过] DDNS_SSH_helper.sh 正在运行" >> /tmp/ddns_ssh_helper.log
else
    pkill -f DDNS_SSH_helper.sh 2>/dev/null
    /bin/sh /jffs/scripts/DDNS_SSH_helper.sh >> /tmp/ddns_ssh_helper.log 2>&1 &
    echo $! > "$DDNS_PID"
fi

# ======================= 4.0 防火墙安全与分流逻辑 (完整增强版) =======================
echo_date "======================= 正在部署 WebDAV 安全规则"
# 2525 端口：先清理，再重注入。确保规则在 INPUT 链最前面（防止被 ESTABLISHED 规则抢跑）
while iptables -D INPUT -p tcp --dport 2525 -j ACCEPT 2>/dev/null; do :; done
while iptables -D INPUT -p tcp --dport 2525 -m state --state NEW -m recent --set 2>/dev/null; do :; done
while iptables -D INPUT -p tcp --dport 2525 -m state --state NEW -m recent --update --seconds 60 --hitcount 10 -j DROP 2>/dev/null; do :; done

# 注入顺序：Set -> Update/Drop -> Accept
iptables -I INPUT 1 -p tcp --dport 2525 -j ACCEPT
iptables -I INPUT 1 -p tcp --dport 2525 -m state --state NEW -m recent --update --seconds 60 --hitcount 10 -j DROP
iptables -I INPUT 1 -p tcp --dport 2525 -m state --state NEW -m recent --set

# 8181 IPv6 放行 (常用管理端口)
ip6tables -C INPUT -p tcp --dport 8181 -j ACCEPT 2>/dev/null || ip6tables -I INPUT -p tcp --dport 8181 -j ACCEPT

# AI 分流注入：【增强：自动获取端口与空参保护】
echo_date "======================= 正在注入 AI 助手独立分流规则"

# 1. 动态抓取代理端口，并设置兜底值防止报错
PROXY_PORT=$(nvram get ss_status_proxy_port)
[ -z "$PROXY_PORT" ] && PROXY_PORT=$(nvram get ss_redir_port)
FINAL_PORT=${PROXY_PORT:-3333} # 如果 nvram 全空，强制使用 3333

# 2. 定义作者建议的精准网段 (Cloudflare & X/Twitter)
CF_NETS="104.16.0.0/13 104.24.0.0/14 172.64.0.0/13 131.0.72.0/22"
X_NETS="104.244.42.0/24 192.133.76.0/22 199.16.156.0/22 199.59.148.0/22 204.92.112.0/22 69.195.160.0/19"

# 3. 创建独立 ipset (避免污染 gfwlist)
ipset -N ai_bypass hash:net 2>/dev/null
for net in $CF_NETS $X_NETS; do
    ipset -! add ai_bypass $net 2>/dev/null
done

# 4. 注入分流规则 (增加空参检查和重复清理)
if [ -n "$FINAL_PORT" ]; then
    echo_date "检测到分流端口: $FINAL_PORT"
    # 清理可能存在的旧规则（防止脚本多次运行导致 nat 表堆叠）
    while iptables -t nat -D PREROUTING -p tcp -m set --match-set ai_bypass dst -j REDIRECT --to-ports "$FINAL_PORT" 2>/dev/null; do :; done
    # 注入新规则
    iptables -t nat -I PREROUTING -p tcp -m set --match-set ai_bypass dst -j REDIRECT --to-ports "$FINAL_PORT"
else
    echo_date "错误：未能获取有效端口，跳过 AI 分流注入"
fi

# IPv6 DNS 劫持逻辑：【修正：针对 102.6_0 固件变量为空的适配】
IPV6_LAN=$(nvram get ipv6_lan_addr)
# 如果变量为空，则实时抓取 br0 的 Global 地址
[ -z "$IPV6_LAN" ] && IPV6_LAN=$(ip -6 addr show br0 | grep "global" | awk '{print $2}' | cut -d/ -f1 | head -n1)

if [ -n "$IPV6_LAN" ]; then
    echo_date "检测到 IPv6: $IPV6_LAN，执行【带标记】劫持"
    
    # 【精准清理】：利用 grep 找到带自定义标记的规则并删除
    while ip6tables -t nat -S PREROUTING | grep "CUSTOM_DNS_HIJACK" | sed 's/-A/ip6tables -t nat -D/' | sh 2>/dev/null; do :; done
    
    # 【精准注入】：增加 -m comment 标记，方便作者审计且不破坏原生 ACL
    ip6tables -t nat -I PREROUTING -p udp -m udp --dport 53 -j DNAT --to "$IPV6_LAN" -m comment --comment "CUSTOM_DNS_HIJACK"
else
    echo_date "警告：IPv6 地址抓取失败，请检查拨号状态"
fi

echo_date "======================= FancySS Mod 增强配置部署完成 ====================="
