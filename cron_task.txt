#!/bin/sh

LOG_TIME=$(date '+【%Y%m%d %H:%M:%S】')
echo "$LOG_TIME: ======================= 自定义任务启动开始 ==========================="

# BEGIN CUSTOM CRON TASK
# ======================= 1.0 & 2.0 注入定时守护与清理任务 =======================
for task in frpc_guard rclone_guard clean_logs refresh_ddns rclone_cache_clean ss_update_daily; do
    cru d $task 2>/dev/null
done

# 统一格式注入定时任务
cru a frpc_guard "*/10 * * * * /bin/sh /jffs/scripts/frpc_start.sh >/dev/null 2>&1"
cru a rclone_guard "*/20 * * * * /bin/sh /jffs/scripts/rclone_webdav.sh >/dev/null 2>&1"
cru a refresh_ddns "30 3 */5 * * /jffs/scripts/refresh_ddns.sh >> /jffs/scripts/ddns_refresh.log 2>&1"
cru a ss_update_daily "30 4 * * * sh /jffs/.koolshare/scripts/ss_update.sh update >> /jffs/scripts/ss_update.log 2>&1"
cru a clean_logs "0 3 * * * /jffs/scripts/clean_logs.sh >> /jffs/scripts/clean_logs.txt 2>&1"

# 凌晨 5 点清理并拉起
cru a rclone_cache_clean "0 5 * * * [ -d /tmp/mnt/SF/entware/var/cache/rclone ] && rm -rf /tmp/mnt/SF/entware/var/cache/rclone/* && /bin/sh /jffs/scripts/rclone_webdav.sh >/dev/null 2>&1"



# ======================= 3.0 后台服务温柔启动 (存在即跳过) =======================
start_service_soft() {
    local _script="$1"
    local _name=$(basename "$_script")
    
    # 第一重检查：检查 ps 进程列表里是否已经有这个脚本在跑
    if ps -w | grep -v grep | grep -q "$_name"; then
        return 0
    fi
    
    # 直接运行脚本，脚本内部建议也带上同样的“存在即跳过”逻辑
    /bin/sh "$_script" >/dev/null 2>&1 &
}

# 启动服务
start_service_soft "/jffs/scripts/rclone_webdav.sh"
start_service_soft "/jffs/scripts/DDNS_SSH_helper.sh"

# 启动 frpc (内部已带“存在即跳过”逻辑)
[ -f "/jffs/scripts/frpc_start.sh" ] && /bin/sh /jffs/scripts/frpc_start.sh


# 1. 定义彻底清理函数
full_cleanup() {
    local port=$1
    local proto=$2
    # 循环删除所有相关的自定义规则，直到该端口的所有匹配项都消失
    while $proto -D INPUT -p tcp --dport $port -j ACCEPT 2>/dev/null; do :; done
    while $proto -D INPUT -p tcp --dport $port -m state --state NEW -m recent --set 2>/dev/null; do :; done
    while $proto -D INPUT -p tcp --dport $port -m state --state NEW -m recent --rcheck --seconds 60 --hitcount 8 -j DROP 2>/dev/null; do :; done
    while $proto -D INPUT -p tcp --dport $port -m state --state NEW -m recent --rcheck --seconds 60 --hitcount 20 -j DROP 2>/dev/null; do :; done
    while $proto -D INPUT -p tcp --dport $port -m state --state NEW -m recent --update --seconds 60 --hitcount 8 -j LOG --log-prefix "${port}-BLOCK: " 2>/dev/null; do :; done
}

# 2. 对 2525 和 8181 执行 IPv4/IPv6 双栈清理
for p in 2525 8181; do
    full_cleanup $p iptables
    full_cleanup $p ip6tables
done

# 3. 重新添加“无条件 ACCEPT”规则（最基础的通行证）
iptables -I INPUT -p tcp --dport 2525 -j ACCEPT
ip6tables -I INPUT -p tcp --dport 2525 -j ACCEPT
iptables -I INPUT -p tcp --dport 8181 -j ACCEPT
ip6tables -I INPUT -p tcp --dport 8181 -j ACCEPT

# 4. 彻底清空内核中的 recent 模块内存快照
if [ -f /proc/net/xt_recent/DEFAULT ]; then
    echo / > /proc/net/xt_recent/DEFAULT
fi

echo "$LOG_TIME: ======================= 自定义任务启动完毕 =========================== "
